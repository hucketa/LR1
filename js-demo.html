function calculate() {
const num1 = parseFloat(document.getElementById('num1').value);
const num2 = parseFloat(document.getElementById('num2').value);
const operator = document.getElementById('operator').value;
let result;
switch(operator) {
case '+': result = num1 + num2; break;
case '-': result = num1 - num2; break;
case '*': result = num1 * num2; break;
case '/': result = num1 / num2; break;
}
document.getElementById('calcResult').textContent =
`Результат: ${result}`;
}

function analyzeText() {
const text = document.getElementById('textInput').value;
const length = text.length;
const words = text.split(' ').length;

const lower = text.toLowerCase();

<p>Символів: ${length}</p>
<p>Слів: ${words}</p>
<p>Великі літери: ${upper}</p>
<p>Малі літери: ${lower}</p>
`;
}

const list = document.getElementById('dynamicList');

const li = document.createElement('li');

li.onclick = function() {

    function checkAge() {
const age = parseInt(document.getElementById('ageInput').value);
const result = document.getElementById('ageResult');
if (age < 0 || isNaN(age)) {
result.innerHTML = '❌ Введіть коректний вік';
result.className = 'error-message';
} else if (age < 18) {
result.innerHTML = ' Ви неповнолітній';
result.className = 'info-message';
} else if (age < 65) {
result.innerHTML = ' Ви дорослий';
result.className = 'success-message';
} else {
result.innerHTML = ' Ви пенсіонер';
result.className = 'info-message';
}
}

    Генератор таблиці множення

    Показати код 
    function generateTable() {
const num = parseInt(document.getElementById('tableNumber').value);
let html = '<table class="multiplication-table">';
for (let i = 1; i <= 10; i++) {
const result = num * i;
html += `
<tr>
<td>${num} × ${i}</td>
<td>=</td>
<td>${result}</td>
</tr>
`;
}
html += '</table>';
document.getElementById('tableResult').innerHTML = html;
}
